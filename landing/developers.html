hardhat.config.ts
// hardhat.config.ts import type { HardhatUserConfig } from "hardhat/config"; const config: HardhatUserConfig = { solidity: "0.8.24", networks: { chakram: { url: "https://rpc.chakram.one", chainId: 11640, accounts: [process.env.DEPLOYER_KEY!], }, }, }; export default config;
foundry.toml
# foundry.toml [profile.default] src = "src" out = "out" libs = ["lib"] [rpc_endpoints] chakram = "https://rpc.chakram.one" # Deploy: # forge script script/Deploy.s.sol \ # --rpc-url chakram \ # --broadcast \ # --private-key $DEPLOYER_KEY
chains.ts
import { createPublicClient, http, defineChain } from "viem"; export const chakram = defineChain({ id: 11640, name: "Chakram", nativeCurrency: { name: "Chakram", symbol: "CHK", decimals: 18 }, rpcUrls: { default: { http: ["https://rpc.chakram.one"] }, }, blockExplorers: { default: { name: "Chakram Explorer", url: "https://explorer.chakram.one", }, }, }); const client = createPublicClient({ chain: chakram, transport: http(), }); const block = await client.getBlockNumber();
provider.ts
import { ethers } from "ethers"; const provider = new ethers.JsonRpcProvider( "https://rpc.chakram.one" ); const network = await provider.getNetwork(); // { chainId: 11640n, name: "unknown" } const block = await provider.getBlock("latest"); console.log(block?.number, block?.hash);

06 FAQ

Asked.
Answered.

Is Chakram EVM-compatible?
Yes. Chakram runs go-ethereum v1.17 as its execution layer — unmodified. Solidity contracts, Hardhat, Foundry, viem, ethers.js, and wallets such as MetaMask work exactly as they do on Ethereum. No code changes are required.
What are the chain ID and RPC?
Chain ID: 11640 (hex 0x2D78). Public RPC: https://rpc.chakram.one. No API key required.
How fast is finality?
Single-block finality, roughly every second. CometBFT's PBFT-style consensus requires >⅔ of validators to sign a block before it is committed. Once committed, the block is irreversible — there are no reorgs on Chakram, ever.
How do I add Chakram to MetaMask?
Click Add Chakram to wallet anywhere on this page — it calls wallet_addEthereumChain and fills in all the details automatically. Alternatively, add it manually: Network name Chakram, RPC URL https://rpc.chakram.one, Chain ID 11640, symbol CHK.
Can I deploy OpenZeppelin contracts?
Yes, without changes. ERC-20, ERC-721, ERC-1155, Governor, Ownable — everything from the OpenZeppelin Contracts library works as-is. The EVM and its opcode set are identical to Ethereum's.
Is there a faucet or testnet CHK?
A public faucet is coming. In the meantime, validator nodes running in dev mode expose a faucet at node.chakram.one — Wallet tab → Faucet. You can also reach out to the Chakram team on GitHub to request testnet CHK directly.
Where can I verify contracts?
The block explorer at explorer.chakram.one has built-in contract verification. Navigate to the contract address → Contract tab → submit source code, compiler version, and optimizer settings.
Is Chakram a Layer 2?
No. Chakram is a sovereign Layer 1 blockchain. It is not built on Ethereum, Avalanche, or any other base chain. CometBFT provides its own BFT consensus; go-ethereum provides EVM execution. The network is fully independent.
